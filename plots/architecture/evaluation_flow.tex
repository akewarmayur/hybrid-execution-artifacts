% ============================================================
% TOP ROW: one aligned step
% ============================================================

\node[inputbox, text width=.16\linewidth] (input) {
\textbf{Aligned step $t$}\\[-0.2mm]
reference $r$\\
restart candidate $c$
};

\node[metricbox, text width=.30\linewidth, right=of input] (metrics) {
\textbf{Compute deviations}\\[-0.2mm]
objective $\Delta J_t$,\;
distribution $H_t$\\
gradient $G_t$\\[-0.8mm]
{\scriptsize $H_t:~0=$ same,\; $1=$ disjoint}
};

\node[passbox, text width=.20\linewidth, right=of metrics] (gate) {
\textbf{Check\\calibrated limits}\\[-0.2mm]
$\Delta J_t\le\tau_J,$

 $H_t\le\tau_H,\quad
 G_t\le\tau_G$
};

\node[passbox, text width=.18\linewidth, right=of gate] (decision) {
\textbf{Step decision}\\[-0.2mm]
$I_t=1$ if all pass\\
$I_t=0$ otherwise
};

\draw[arrow] (input) -- (metrics);
\draw[arrow] (metrics) -- (gate);
\draw[arrow] (gate) -- (decision);


% ============================================================
% WIDE, SHALLOW CALIBRATION BOX
% ============================================================

\node[
  calibbox,
  text width=.62\linewidth,
  minimum height=7mm,
  above=3mm of gate,
   xshift=-8mm
] (calib) {
\textbf{Independent calibration:}
uninterrupted runs determine
$\tau_J,\;\tau_H,\;\tau_G$
};

\draw[arrow] (calib.south -| gate.north) -- (gate.north);


% ============================================================
% BOTTOM ROW: use full width from the left
% ============================================================

\node[
  seqbox,
  text width=.30\linewidth,
  at={($(input.south west)+(0,-9mm)$)},
  anchor=north west
] (sequence) {
\textbf{Collect $B$ aligned decisions}\\[-0.2mm]
$I_0,\;I_1,\ldots,I_{B-1}$
};

\node[
  trajbox,
  text width=.42\linewidth,
  right=4mm of sequence
] (trajectory) {
\textbf{Trajectory criterion}\\[-0.2mm]
final step: $C=I_{B-1}$\\
stable window: $S=$ at least $W$ consecutive $I_t=1$
};

\node[
  goodbox,
  text width=.10\linewidth,
  right=7mm of trajectory,
  yshift=4mm
] (accepted) {
\textbf{Accepted $\checkmark$}\\[-0.2mm]
$C\land S$
};

\node[
  badbox,
  text width=.10\linewidth,
  right=7mm of trajectory,
  yshift=-3mm
] (unsafe) {
\textbf{Unsafe $\times$}\\[-0.2mm]
otherwise
};


% ============================================================
% ROUTING
% ============================================================

\draw[arrow]
  (decision.south)

  -|
  node[pos=.40, below, font=\scriptsize]
       {repeat for $t=0,\ldots,B-1$}
  (sequence.north);

\draw[arrow] (sequence) -- (trajectory);

\draw[arrow]
  (trajectory.east)
  -- ++(3mm,0)
  |-
  node[pos=.72, above, font=\scriptsize] {yes}
  (accepted.west);

\draw[arrow]
  (trajectory.east)
  -- ++(3mm,0)
  |-
  node[pos=.72, below, font=\scriptsize] {no}
  (unsafe.west);

\caption{Continuation-outcome evaluation. Each aligned restart step is compared
with its uninterrupted reference using objective, distributional, and gradient
deviations. Independently calibrated limits convert these deviations into
per-step decisions $I_t$, which are aggregated across the $B$-step horizon
using the final-step condition $C$ and stable-window condition $S$.}
\label{fig:continuation-envelope}
